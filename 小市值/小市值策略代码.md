'''
筛选出PE(TTM)小于20的股票，从其中市值最小和最大各取一半（共30只），
每5个交易日调仓一次，开盘买入。
'''

## 初始化函数，设定要操作的股票、基准等等
def initialize(context):
    # 设定沪深300作为基准
    set_benchmark('000300.XSHG')
    # True为开启动态复权模式，使用真实价格交易
    set_option('use_real_price', True) 
    # 设定成交量比例
    set_option('order_volume_ratio', 1)
    # 股票类交易手续费是：买入时佣金万分之三，卖出时佣金万分之三加千分之一印花税, 每笔交易佣金最低扣5块钱
    set_order_cost(OrderCost(open_tax=0, close_tax=0.001, \
                             open_commission=0.0003, close_commission=0.0003,\
                             close_today_commission=0, min_commission=5), type='stock')
    # 持仓数量（市值最大/最小各取一半，各 g.stocknum//2 只）
    g.stocknum = 30 
    # 交易日计时器
    g.days = 0 
    # 调仓频率
    g.refresh_rate = 5
    # 次新股过滤天数（上市不足180天的股票剔除）
    g.new_stock_limit = 180
    # 流动性约束：日换手率下限（%），过滤僵尸股（低于1%的股票成交惨淡，买卖冲击成本高）
    # 注意：不用 valuation.turnover_ratio（该字段在聚宽返回NaN/历史数据不可靠），
    # 改用"昨日成交量/流通股本"自算换手率
    g.min_turnover = 1
    # 运行函数
    run_daily(trade, 'every_bar')

## 选出股票
def check_stocks(context):
    # 设定查询条件：PE(TTM)小于20（pe_ratio>0排除亏损股，负PE永远小于20），不限市值区间
    # 只查 PE 和流通股本，换手率在 Python 层用昨日成交量自算（valuation.turnover_ratio 不可靠）
    q = query(
            valuation.code,
            valuation.circulating_cap,  # 流通股本（万股）
            valuation.pe_ratio
        ).filter(
            valuation.pe_ratio > 0,
            valuation.pe_ratio < 20
        ).order_by(
            valuation.market_cap.desc()
        )

    # 显式锚定前一交易日（get_fundamentals 默认取"前一自然日"，周一/节假日会取到非交易日导致数据缺失）
    df = get_fundamentals(q, date=context.previous_date)
    if df.empty:
        return []

    codes = list(df['code'])
    # 昨日成交量（股）：attribute_history 不支持传 list（会报错），改用 get_price 批量取
    # end_date 锚定前一交易日，无未来函数
    vols = {}
    batch_size = 100
    for i in range(0, len(codes), batch_size):
        price = get_price(codes[i:i+batch_size], count=1,
                          end_date=context.previous_date,
                          fields=['volume'], frequency='daily', panel=False)
        # 兼容两种返回格式：长表(code/volume列) 或 宽表(MultiIndex)
        if 'code' in price.columns:
            vols.update(dict(zip(price['code'], price['volume'])))
        else:
            vols.update(price['volume'].iloc[0].to_dict())
    df = df.set_index('code')
    df['volume'] = df.index.map(lambda c: vols.get(c, 0))

    # 换手率(%) = 昨日成交量(股) / (流通股本(万股) * 10000) * 100
    df['turnover'] = df['volume'] / (df['circulating_cap'] * 10000) * 100

    # 流动性过滤在 Python 层做：换手率不足或数据缺失（NaN/停牌 volume=0）的股票自动剔除
    df = df[df['turnover'] > g.min_turnover]
    buylist = list(df.index)

    # 过滤科创板（688开头）：科创板市价单必须指定保护限价，且波动大，直接剔除
    buylist = [stock for stock in buylist if not stock.startswith('688')]

    # 依次过滤：ST/退市风险、停牌、涨停、次新
    buylist = filter_st_stock(buylist)
    buylist = filter_paused_stock(buylist)
    buylist = filter_limit_up_stock(buylist)
    buylist = filter_new_stock(context, buylist)

    # 各取一半：buylist 按市值降序排列（market_cap.desc），
    # 取前 half 只（市值最大）+ 后 half 只（市值最小）
    half = g.stocknum // 2
    if len(buylist) <= g.stocknum:
        return buylist  # 池子不足30只时全要
    return buylist[:half] + buylist[-half:]

## 交易函数
def trade(context):
    if g.days % g.refresh_rate != 0:
        g.days += 1
        return

    ## 选股
    stock_list = check_stocks(context)

    ## 每只股票的等权目标价值（用总权益而非现金，避免卖出未成交时现金不足）
    target_value = context.portfolio.total_value / g.stocknum

    ## 卖出不在目标名单中的持仓
    for stock in list(context.portfolio.positions.keys()):
        if stock not in stock_list:
            order_target_value(stock, 0)

    ## 将名单内股票调整至目标价值（已持有且未超配的股票不会先卖再买，减少摩擦成本）
    ## 容差判断：持仓市值与目标价值差异不足一手(100股)金额时跳过，
    ## 避免"需减仓但不足一手"导致订单委托失败（A股卖出必须是100股整数倍）
    current_data = get_current_data()
    for stock in stock_list:
        if stock not in context.portfolio.positions:
            # 新开仓，直接买入（用 in 判断，聚宽 positions.get 不存在时返回空Position并打WARNING）
            order_target_value(stock, target_value)
            continue
        position = context.portfolio.positions[stock]
        diff = abs(position.value - target_value)
        if diff >= current_data[stock].last_price * 100:
            order_target_value(stock, target_value)

    # 天计数重置
    g.days = 1

# 过滤停牌股票
def filter_paused_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list if not current_data[stock].paused]

# 过滤ST/*ST/退市风险股票
def filter_st_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list
            if not current_data[stock].is_st
            and 'ST' not in current_data[stock].name
            and '*' not in current_data[stock].name
            and '退' not in current_data[stock].name]

# 过滤涨停股票（涨停封板买不进，剔除）
def filter_limit_up_stock(stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list
            if current_data[stock].last_price < current_data[stock].high_limit]

# 过滤次新股（上市不足 g.new_stock_limit 天剔除）
def filter_new_stock(context, stock_list):
    today = context.current_dt.date()
    return [stock for stock in stock_list
            if (today - get_security_info(stock).start_date).days >= g.new_stock_limit]
