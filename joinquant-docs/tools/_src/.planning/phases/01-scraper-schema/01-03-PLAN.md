3. Create `scraper/api_docs.py`:

   ```python
   import time
   import random
   from bs4 import BeautifulSoup
   from playwright.sync_api import BrowserContext

   API_SECTIONS = [
       {"url": "https://www.joinquant.com/help/api/help#Stock:获取股票数据", "section": "获取股票数据"},
       {"url": "https://www.joinquant.com/help/api/help#Stock:获取单季度年度财务数据", "section": "获取单季度年度财务数据"},
       {"url": "https://www.joinquant.com/help/api/help#Stock:上市公司概况", "section": "上市公司概况"},
       {"url": "https://www.joinquant.com/help/api/help#Stock:获取融资融券标的列表", "section": "获取融资融券标的列表"},
   ]

   def scrape_all_api_sections(context: BrowserContext) -> list[dict]:
       """Scrape all 4 API doc pages, return list of function records."""
       all_records = []
       page = context.new_page()
       try:
           for i, section_info in enumerate(API_SECTIONS):
               if i > 0:
                   time.sleep(random.uniform(2.0, 3.0))  # polite delay
               records = scrape_api_section(page, section_info["url"], section_info["section"])
               all_records.extend(records)
               if len(records) == 0:
                   print(f"WARNING: 0 functions found on {section_info['section']} -- check selectors")
       finally:
           page.close()
       return all_records

   def scrape_api_section(page, url: str, section: str) -> list[dict]:
       """Scrape a single API doc page. Returns list of function record dicts."""
       page.goto(url, timeout=30000)
       page.wait_for_selector("[CONTENT_LOADED_SELECTOR from SELECTORS.md]", timeout=20000)
       html = page.content()
       return _parse_api_functions(html, section)

   def _parse_api_functions(html: str, section: str = "") -> list[dict]:
       """Parse API function blocks from HTML. Pure function, no browser needed."""
       soup = BeautifulSoup(html, "lxml")
       records = []
       for func_block in soup.select("[FUNCTION_WRAPPER_SELECTOR from SELECTORS.md]"):
           name_el = func_block.select_one("[FUNCTION_NAME_SELECTOR]")
           if not name_el:
               continue
           record = {
               "function_name": name_el.get_text(strip=True),
               "section": section,
               "call_signature": _extract_text(func_block, "[CALL_SIGNATURE_SELECTOR]"),
               "description": _extract_text(func_block, "[DESCRIPTION_SELECTOR]"),
               "return_type": _extract_text(func_block, "[RETURN_TYPE_SELECTOR]"),
               "example_code": _extract_text(func_block, "[EXAMPLE_CODE_SELECTOR]"),
               "params": _extract_params(func_block),
               "return_attrs": _extract_return_attrs(func_block),
           }
           records.append(record)
       return records

   def _extract_text(soup_el, selector: str) -> str | None:
       el = soup_el.select_one(selector)
       return el.get_text(strip=True) if el else None

   def _extract_params(func_block) -> list[dict]:
       params = []
       for row in func_block.select("[PARAM_ROW_SELECTOR from SELECTORS.md]"):
           cells = row.select("td")
           if len(cells) >= 2:
               params.append({
                   "param_name": cells[0].get_text(strip=True),
                   "param_type": cells[1].get_text(strip=True) if len(cells) > 1 else None,
                   "description": cells[2].get_text(strip=True) if len(cells) > 2 else None,
               })
       return params

   def _extract_return_attrs(func_block) -> list[dict]:
       attrs = []
       for row in func_block.select("[RETURN_ATTR_ROW_SELECTOR from SELECTORS.md]"):
           cells = row.select("td")
           if len(cells) >= 1:
               attrs.append({
                   "attr_name": cells[0].get_text(strip=True),
                   "attr_type": cells[1].get_text(strip=True) if len(cells) > 1 else None,
                   "description": cells[2].get_text(strip=True) if len(cells) > 2 else None,
               })
       return attrs
   ```
