```python
import time
import random
from playwright.sync_api import BrowserContext
from rich import print as rprint

def scrape_all_strategies(context: BrowserContext) -> list[dict]:
    """Scrape all strategies from the 经典策略学习 sidebar. Requires authenticated session."""
    page = context.new_page()
    records = []
    try:
        # Navigate to the strategy learning section
        page.goto("https://www.joinquant.com/algorithm/index/list", timeout=30000)
        # OR the actual strategy learning URL discovered in SELECTORS.md
        page.wait_for_selector("[SIDEBAR_SELECTOR from SELECTORS.md]", timeout=20000)

        # Get all category items from sidebar
        categories = _get_sidebar_categories(page)
        rprint(f"Found {len(categories)} strategy categories")

        for cat_name, strategy_links in categories.items():
            for link_info in strategy_links:
                time.sleep(random.uniform(2.0, 3.0))  # polite delay
                record = _extract_strategy(page, link_info, cat_name)
                if record:
                    records.append(record)
                    rprint(f"  Extracted: {record['name']} ({cat_name})")

        if len(records) == 0:
            rprint("[bold red]WARNING: 0 strategies found -- check selectors[/bold red]")
    finally:
        page.close()
    return records

def _get_sidebar_categories(page) -> dict[str, list[dict]]:
    """Parse the sidebar navigation to get category -> strategy links mapping."""
    categories = {}
    # Use real selectors from SELECTORS.md
    for cat_el in page.query_selector_all("[CATEGORY_ITEM_SELECTOR from SELECTORS.md]"):
        cat_name = cat_el.inner_text().strip()
        # Click category to expand if needed
        cat_el.click()
        time.sleep(0.5)  # Brief wait for expansion animation
        strategy_els = cat_el.query_selector_all("[STRATEGY_LINK_SELECTOR from SELECTORS.md]")
        links = []
        for s in strategy_els:
            links.append({
                "name": s.inner_text().strip(),
                "element": s,
                "href": s.get_attribute("href"),
            })
        categories[cat_name] = links
    return categories

def _extract_strategy(page, link_info: dict, category: str) -> dict | None:
    """Navigate to a strategy page and extract its content."""
    try:
        # Try direct URL navigation first (per user decision)
        if link_info.get("href"):
            page.goto(f"https://www.joinquant.com{link_info['href']}", timeout=30000)
        else:
            # Fall back to clicking the sidebar link
            link_info["element"].click()

        # Wait for strategy code content to load
        page.wait_for_selector("[CODE_CONTENT_SELECTOR from SELECTORS.md]", timeout=15000)

        code_el = page.query_selector("[CODE_CONTENT_SELECTOR from SELECTORS.md]")
        desc_el = page.query_selector("[DESCRIPTION_SELECTOR from SELECTORS.md]")

        return {
            "name": link_info["name"],
            "category": category,
            "code_content": code_el.inner_text() if code_el else "",
            "description": desc_el.inner_text().strip() if desc_el else None,
        }
    except Exception as e:
        rprint(f"[yellow]WARNING: Failed to extract strategy '{link_info['name']}': {e}[/yellow]")
        return {
            "name": link_info["name"],
            "category": category,
            "code_content": "",
            "description": None,
        }
```

Create `run_scrape.py` as the single CLI entry point that orchestrates the full ingestion pipeline:

```python
#!/usr/bin/env python3
"""JoinQuant Knowledge Base Scraper - Full ingestion pipeline.

Usage: uv run python run_scrape.py
"""

import sqlite3
from pathlib import Path
from playwright.sync_api import sync_playwright
from rich import print as rprint
from rich.console import Console

from auth import ensure_authenticated, AUTH_FILE
from scraper.api_docs import scrape_all_api_sections
from scraper.strategies import scrape_all_strategies
from db.schema import init_db
from db.seed import upsert_api_doc, upsert_api_params, upsert_api_return_attrs, upsert_strategy

DB_FILE = Path("jq_knowledge.db")
console = Console()

def count_rows(conn: sqlite3.Connection, table: str) -> int:
    return conn.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]

def main():
    rprint("[bold blue]JoinQuant Knowledge Base Scraper[/bold blue]")
    rprint(f"Database: {DB_FILE}")

    # Step 1: Initialize database
    conn = init_db(DB_FILE)
    rprint("[green]Database initialized[/green]")

    with sync_playwright() as p:
        # Step 2: Launch browser, load session if available
        storage = str(AUTH_FILE) if AUTH_FILE.exists() else None
        browser = p.chromium.launch(headless=True)
        context = browser.new_context(storage_state=storage)

        # Step 3: Ensure authenticated session
        rprint("Checking authentication...")
        ensure_authenticated(context)
        rprint("[green]Authenticated[/green]")

        # Step 4: Scrape API docs (4 pages)
        rprint("\n[bold]Scraping API documentation...[/bold]")
        api_records = scrape_all_api_sections(context)
        rprint(f"Extracted {len(api_records)} API functions")

        for record in api_records:
            upsert_api_doc(conn, {
                "function_name": record["function_name"],
                "section": record["section"],
                "call_signature": record.get("call_signature"),
                "description": record.get("description"),
                "return_type": record.get("return_type"),
                "example_code": record.get("example_code"),
            })
            if record.get("params"):
                upsert_api_params(conn, record["function_name"], record["params"])
            if record.get("return_attrs"):
                upsert_api_return_attrs(conn, record["function_name"], record["return_attrs"])

        # Step 5: Scrape strategies (requires auth)
        rprint("\n[bold]Scraping classic strategies...[/bold]")
        strategy_records = scrape_all_strategies(context)
        rprint(f"Extracted {len(strategy_records)} strategies")

        for record in strategy_records:
            upsert_strategy(conn, record)

        browser.close()

    # Step 6: Print summary
    rprint("\n[bold green]Scraping complete![/bold green]")
    rprint(f"  api_docs:        {count_rows(conn, 'api_docs')} rows")
    rprint(f"  api_params:      {count_rows(conn, 'api_params')} rows")
    rprint(f"  api_return_attrs:{count_rows(conn, 'api_return_attrs')} rows")
    rprint(f"  strategies:      {count_rows(conn, 'strategies')} rows")

    # Sanity checks
    for table in ["api_docs", "strategies"]:
        if count_rows(conn, table) == 0:
            rprint(f"[bold red]WARNING: {table} has 0 rows -- something went wrong![/bold red]")

    conn.close()

if __name__ == "__main__":
    main()
```
