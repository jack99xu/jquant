**Part B: Build auth.py**

Create `auth.py` using the selectors discovered in Part A:

```python
import os
from pathlib import Path
from dotenv import load_dotenv
from playwright.sync_api import BrowserContext

load_dotenv()

AUTH_FILE = Path("auth.json")

def ensure_authenticated(context: BrowserContext) -> None:
    """Load existing session or perform fresh login. Saves auth.json on success."""
    page = context.new_page()
    try:
        # Navigate to an authenticated page to test session
        page.goto("https://www.joinquant.com/algorithm/index/list", timeout=15000)
        page.wait_for_load_state("domcontentloaded")

        # If redirected to login or login form visible, session is invalid
        if "login" in page.url or page.query_selector("[DISCOVERED_LOGIN_FORM_SELECTOR]") is not None:
            _do_login(page)
            context.storage_state(path=str(AUTH_FILE))
    finally:
        page.close()

def _do_login(page) -> None:
    """Fill and submit JoinQuant login form. Fails fast on error."""
    page.goto("https://www.joinquant.com/login", timeout=15000)
    page.wait_for_selector("[DISCOVERED_USERNAME_SELECTOR]", timeout=10000)

    username = os.environ.get("JQ_USERNAME")
    password = os.environ.get("JQ_PASSWORD")
    if not username or not password:
        raise RuntimeError("JQ_USERNAME and JQ_PASSWORD must be set in .env")

    page.fill("[DISCOVERED_USERNAME_SELECTOR]", username)
    page.fill("[DISCOVERED_PASSWORD_SELECTOR]", password)
    page.click("[DISCOVERED_SUBMIT_SELECTOR]")

    # Wait for navigation away from login page
    page.wait_for_url("**/algorithm/**", timeout=30000)
```

Replace all `[DISCOVERED_*_SELECTOR]` placeholders with the actual selectors found during the headed discovery spike.

The auth module must:
- Load credentials from .env via python-dotenv
- Check session validity before attempting login
- Save auth.json after successful login
- Fail fast with clear error if login fails (no retry loop)
- Respect the 2-3 second polite delay pattern
